  <script>
    let snapshotInicial = {};

    // Iniciar carga al abrir el modal
    window.addEventListener('DOMContentLoaded', () => {
      google.script.run.withSuccessHandler(llenarFormulario).obtenerConfiguracionActual();
    });

    function capturarEstadoActual() {
      return {
        MODO_PRUEBA: document.getElementById("MODO_PRUEBA").checked,
        EMAIL_PRUEBA: document.getElementById("EMAIL_PRUEBA").value.trim(),
        DIAS_CARENCIA_RESOLUCION: document.getElementById("DIAS_CARENCIA_RESOLUCION").value,
        DIAS_CICLO_SEGUIMIENTO: document.getElementById("DIAS_CICLO_SEGUIMIENTO").value,
        DIAS_CICLO_TRAMITES: document.getElementById("DIAS_CICLO_TRAMITES").value,
        COLOR_ORO: document.getElementById("COLOR_ORO").value.toUpperCase(),
        COLOR_FONDO: document.getElementById("COLOR_FONDO").value.toUpperCase(),
        COLOR_TEXTO: document.getElementById("COLOR_TEXTO").value.toUpperCase(),
        URL_LOGO_PUBLICO: document.getElementById("URL_LOGO_PUBLICO").value.trim()
      };
    }

    function llenarFormulario(datos) {
      document.getElementById("MODO_PRUEBA").checked = !!datos.MODO_PRUEBA;
      document.getElementById("EMAIL_PRUEBA").value = datos.EMAIL_PRUEBA || "";
      document.getElementById("DIAS_CARENCIA_RESOLUCION").value = datos.DIAS_CARENCIA_RESOLUCION || 60;
      document.getElementById("DIAS_CICLO_SEGUIMIENTO").value = datos.DIAS_CICLO_SEGUIMIENTO || 30;
      document.getElementById("DIAS_CICLO_TRAMITES").value = datos.DIAS_CICLO_TRAMITES || 30;
      document.getElementById("COLOR_ORO").value = datos.COLOR_ORO || "#D4AF37";
      document.getElementById("COLOR_FONDO").value = datos.COLOR_FONDO || "#F7F7F7";
      document.getElementById("COLOR_TEXTO").value = datos.COLOR_TEXTO || "#333333";
      document.getElementById("URL_LOGO_PUBLICO").value = datos.URL_LOGO_PUBLICO || "";

      // Tomamos la "fotografía" inicial
      snapshotInicial = capturarEstadoActual();

      // Ocultamos el velo de carga con efecto fade-out
      const overlay = document.getElementById("loading-overlay");
      overlay.style.opacity = "0";
      setTimeout(() => { overlay.style.display = "none"; }, 300);

      // Asignamos detectores de eventos a todos los controles
      const inputs = document.querySelectorAll("input");
      inputs.forEach(input => {
        input.addEventListener("input", verificarCambios);
        input.addEventListener("change", verificarCambios);
      });
    }

    function verificarCambios() {
      const estadoActual = capturarEstadoActual();
      let isDirty = false;

      for (let key in snapshotInicial) {
        if (estadoActual[key] !== snapshotInicial[key]) {
          isDirty = true;
          break;
        }
      }

      const btn = document.getElementById("btnGuardar");
      btn.disabled = !isDirty;
    }

    function guardar() {
      const btn = document.getElementById("btnGuardar");
      btn.disabled = true;
      btn.innerText = "Guardando...";

      const datos = capturarEstadoActual();

      google.script.run.withSuccessHandler(() => {
        // Actualizamos el snapshot con los nuevos valores guardados
        snapshotInicial = Object.assign({}, datos);
        btn.innerText = "Guardar Configuración";
        btn.disabled = true;

        const alerta = document.getElementById("alerta");
        alerta.style.display = "block";
        setTimeout(() => { alerta.style.display = "none"; }, 2500);
      }).guardarConfiguracion(datos);
    }
  </script>
