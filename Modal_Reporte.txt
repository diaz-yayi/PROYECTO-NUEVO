  <script>
    // Recuperamos los datos enviados desde el .gs
    const data = JSON.parse(<?= datos ?>);
    const div = document.getElementById("contenido");
    const icono = document.getElementById("icono-estado");
    let html = "";

    if (data.prueba) {
      html += `<div class="prueba">⚠️ EL MODO PRUEBA ESTÁ ACTIVADO<br>La base de datos no fue modificada.</div>`;
    }

    if (data.pausado) {
      icono.innerText = "⏳";
      html += `<div class="alerta-pausa">¡Base de datos masiva detectada!<br><br>Para proteger los servidores, el sistema ha pausado su ejecución temporalmente. Despertará y continuará de forma automática en 1 minuto en segundo plano. Ya puedes cerrar esta ventana.</div>`;
    } else {
      if (data.tipo === 'auditoria') {
        html += `<p>El <strong>Trabajador Nocturno</strong> ha terminado de analizar los tiempos de todos los expedientes.</p>`;
        if(data.procesados > 0) {
          html += `<p>Se detectaron y marcaron <strong>${data.procesados}</strong> expediente(s) listos para recibir notificación.</p>`;
        } else {
          html += `<p>No se encontraron expedientes vencidos en este momento.</p>`;
        }
      } 
      else if (data.tipo === 'envio') {
        if (data.prueba) {
          if (data.simulacro > 0) {
            html += `<p>Si esto fuera una ejecución real, se habrían enviado <strong>${data.simulacro}</strong> correos distribuidos de la siguiente forma:</p><ul>`;
            for(let est in data.desglose) {
              html += `<li><strong>${est}</strong>: ${data.desglose[est]} correo(s)</li>`;
            }
            html += `</ul><p><em>* Revisa la bandeja de <strong>${data.emailDestino}</strong> para ver las muestras visuales y el reporte detallado.</em></p>`;
          } else {
            html += `<p>No hay correos pendientes de enviar hoy.</p>`;
          }
        } else {
          html += `<p>Se han enviado <strong>${data.enviados}</strong> correos oficiales a los clientes reales.</p>`;
          html += `<p>Tu base de datos ha sido actualizada y la fecha de última notificación fue registrada.</p>`;
        }
      }
    }

    div.innerHTML = html;
  </script>
