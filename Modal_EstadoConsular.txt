  <script>
    let candidatos = [];
    let totalCandidatos = 0;
    let currentIndex = 0;
    let isProcessing = false;
    let cancelRequested = false;

    let stats = {
      consultados: 0,
      actualizados: 0,
      noEncontrados: 0,
      errores: 0
    };

    window.addEventListener('beforeunload', function (e) {
      if (isProcessing && !cancelRequested) {
        e.preventDefault();
        e.returnValue = 'Hay una verificación en curso. ¿Deseas detenerla o pasarla a segundo plano?';
        return e.returnValue;
      }
    });

    window.onload = function() {
      google.script.run
        .withSuccessHandler(onEscaneoComplete)
        .withFailureHandler(onErrorEscaneo)
        .obtenerListaExpedientesLaHabana();
    };

    function onEscaneoComplete(res) {
      document.getElementById('start-loading').style.display = 'none';
      document.getElementById('start-content').style.display = 'block';

      if (!res.ok) {
        alert(res.mensaje || 'Error al cargar expedientes.');
        return;
      }

      candidatos = res.candidatos || [];
      totalCandidatos = candidatos.length;

      document.getElementById('txt-total-exp').innerText = totalCandidatos;

      if (res.modoPrueba) {
        document.getElementById('badge-modo-prueba').style.display = 'block';
      }

      if (totalCandidatos === 0) {
        document.getElementById('btn-iniciar').disabled = true;
        document.getElementById('btn-iniciar').innerText = 'No hay expedientes pendientes';
      }
    }

    function onErrorEscaneo(err) {
      document.getElementById('start-loading').innerText = 'Error de comunicación: ' + err.message;
    }

    function switchScreen(screenId) {
      document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
      document.getElementById(screenId).classList.add('active');
    }

    function logConsole(msg, type) {
      const consoleBox = document.getElementById('console');
      const div = document.createElement('div');
      div.className = 'console-line ' + (type || '');
      div.innerText = msg;
      consoleBox.appendChild(div);
      consoleBox.scrollTop = consoleBox.scrollHeight;
    }

    function iniciarVerificacion() {
      if (totalCandidatos === 0) return;
      isProcessing = true;
      switchScreen('screen-progress');
      currentIndex = 0;
      cancelRequested = false;
      procesarSiguiente();
    }

    function detenerYSalir() {
      cancelRequested = true;
      isProcessing = false;
      document.getElementById('btn-cancelar').disabled = true;
      document.getElementById('btn-segundo-plano').disabled = true;
      logConsole('⚠️ Deteniendo verificación al terminar expediente en curso...', 'warn');
    }

    function pasarASegundoPlano() {
      cancelRequested = true;
      isProcessing = false;

      const currentItem = candidatos[currentIndex] || candidatos[0];
      const startFromRow = currentItem ? currentItem.rowIndex : 1;

      logConsole('🔄 Transfiriendo verificación restante al servidor en segundo plano...', 'info');

      document.getElementById('btn-cancelar').disabled = true;
      document.getElementById('btn-segundo-plano').innerText = '⏳ Transfiriendo...';
      document.getElementById('btn-segundo-plano').disabled = true;

      google.script.run
        .withSuccessHandler(function() {
          alert('✅ La verificación continuará en segundo plano automáticamente. Puedes cerrar la ventana.');
          google.script.host.close();
        })
        .withFailureHandler(function(err) {
          alert('Error al transferir a segundo plano: ' + err.message);
        })
        .delegarASegundoPlano(startFromRow);
    }

    function procesarSiguiente() {
      if (cancelRequested || currentIndex >= totalCandidatos) {
        isProcessing = false;
        finalizarProceso();
        return;
      }

      const item = candidatos[currentIndex];
      const numActual = currentIndex + 1;
      const pct = Math.round((numActual / totalCandidatos) * 100);

      document.getElementById('bar-progress').style.width = pct + '%';
      document.getElementById('txt-progress-counter').innerText = numActual + ' / ' + totalCandidatos + ' (' + pct + '%)';

      logConsole('🌐 [' + numActual + '/' + totalCandidatos + '] Consultando: ' + item.noExpediente + ' — ' + item.nombreCompleto + '...', 'info');

      google.script.run
        .withSuccessHandler(onItemProcesado)
        .withFailureHandler(onItemError)
        .procesarUnExpedienteConsular(item);
    }

    function onItemProcesado(res) {
      stats.consultados++;

      if (res.resultado === 'OK') {
        stats.actualizados++;
        let matchNote = '';
        if (res.varianteUsada && res.varianteUsada !== candidatos[currentIndex].apellidos) {
          matchNote = ' (Coincidencia con: "' + res.varianteUsada + '")';
        }
        logConsole('   ✅ Estado: ' + (res.estadoConsulado || 'OK') + matchNote, 'ok');
      } else if (res.resultado === 'NO_ENCONTRADO') {
        stats.noEncontrados++;
        logConsole('   ⚠️ No encontrado (Verificadas 4 variantes de apellidos).', 'warn');
      } else {
        stats.errores++;
        logConsole('   ❌ Error de conexión: ' + (res.detalle || 'Error API'), 'error');
      }

      currentIndex++;
      procesarSiguiente();
    }

    function onItemError(err) {
      stats.consultados++;
      stats.errores++;
      logConsole('   ❌ Error crítico: ' + err.message, 'error');
      currentIndex++;
      procesarSiguiente();
    }

    function finalizarProceso() {
      isProcessing = false;
      switchScreen('screen-summary');

      if (cancelRequested) {
        document.getElementById('summary-icon').innerText = '⏹';
        document.getElementById('summary-title').innerText = 'Verificación Detenida';
      } else {
        document.getElementById('summary-icon').innerText = '✅';
        document.getElementById('summary-title').innerText = 'Verificación Completada';
      }

      document.getElementById('sum-consultados').innerText = stats.consultados;
      document.getElementById('sum-actualizados').innerText = stats.actualizados;
      document.getElementById('sum-no-encontrados').innerText = stats.noEncontrados;
      document.getElementById('sum-errores').innerText = stats.errores;

      const ahora = new Date().toLocaleString('es-ES');
      document.getElementById('sum-timestamp').innerText = 'Última verificación: ' + ahora;
    }
  </script>
